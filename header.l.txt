
%{
#include <stdio.h>
int macros = 0, headers = 0;
%}

%%
^#define[ \t]+   { macros++; }
^#include[ \t]+  { headers++; }
\n               ;
.                ;
%%

int main()
{
    printf("Enter a C program (Ctrl+Z then Enter to finish):\n");
    yylex();

    printf("No. of macros = %d\n", macros);
    printf("No. of headers = %d\n", headers);

    return 0;
}

    return 1;
