
%{
#include <stdio.h>
int upper = 0, lower = 0;
%}

%%
[A-Z]     { upper++; }
[a-z]     { lower++; }
\n        ;
.         ;
%%

int main()
{
    printf("Enter a string:\n");
    yylex();

    printf("No. of uppercase letters = %d\n", upper);
    printf("No. of lowercase letters = %d\n", lower);

    return 0;
}

int yywrap()
{
    return 1;
}
